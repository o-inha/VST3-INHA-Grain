cmake_minimum_required(VERSION 3.22)
project(INHAGrate VERSION 0.5.0 LANGUAGES CXX)

# Ship a self-contained Windows binary without a Visual C++ redistributable.
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(INHA_CORE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(Dependencies/INHA-Core)
include(Dependencies/INHA-Core/cmake/InhaPlugin.cmake)
option(INHA_PLUGIN_BUILD_VST3 "Build Windows x64 VST3" ${WIN32})
option(INHA_PLUGIN_BUILD_TESTS "Build Grate DSP and module tests" ON)

add_library(GrateEngine STATIC Source/GrateEngine.cpp)
target_include_directories(GrateEngine PUBLIC Source)
target_compile_features(GrateEngine PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(GrateEngine PRIVATE /W4 /permissive- /EHsc /utf-8)
else()
    target_compile_options(GrateEngine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(INHA_PLUGIN_BUILD_VST3)
    if(NOT WIN32)
        message(FATAL_ERROR "The native INHA editor targets Windows.")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Dependencies/VST3_SDK/pluginterfaces/base/ipluginbase.h")
        message(FATAL_ERROR "Fetch Steinberg VST3 SDK v3.7.14_build_55 with submodules into Dependencies/VST3_SDK.")
    endif()
    set(GRATE_PLUGIN_SOURCES Source/NativeVST3.cpp Source/GratePreferences.cpp)
    add_library(INHA_Grate_VST3 MODULE ${GRATE_PLUGIN_SOURCES})
    target_compile_features(INHA_Grate_VST3 PRIVATE cxx_std_17)
    target_include_directories(INHA_Grate_VST3 PRIVATE Source Dependencies/VST3_SDK)
    inha_attach_core(INHA_Grate_VST3)
    target_link_libraries(INHA_Grate_VST3 PRIVATE GrateEngine
        gdi32 gdiplus msimg32 user32 kernel32 ole32 uuid shell32 comctl32 advapi32)
    target_compile_definitions(INHA_Grate_VST3 PRIVATE
        WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(INHA_Grate_VST3 PRIVATE /W4 /permissive- /EHsc /utf-8)
    endif()
    set_target_properties(INHA_Grate_VST3 PROPERTIES PREFIX "" SUFFIX ".vst3" OUTPUT_NAME "INHA Grate")
    set(VST3_BUNDLE "${CMAKE_BINARY_DIR}/dist/INHA Grate.vst3")
    add_custom_command(TARGET INHA_Grate_VST3 POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${VST3_BUNDLE}/Contents/x86_64-win" "${VST3_BUNDLE}/Contents/Resources"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:INHA_Grate_VST3>" "${VST3_BUNDLE}/Contents/x86_64-win/INHA Grate.vst3"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Resources" "${VST3_BUNDLE}/Contents/Resources"
        COMMENT "Assembling INHA Grate Windows x64 VST3 bundle")
endif()

if(INHA_PLUGIN_BUILD_TESTS)
    enable_testing()
    add_executable(GrateEngineTests Tests/GrateEngineTests.cpp)
    target_link_libraries(GrateEngineTests PRIVATE GrateEngine)
    target_compile_features(GrateEngineTests PRIVATE cxx_std_17)
    add_test(NAME GrateEngineTests COMMAND GrateEngineTests)
    add_executable(GrateAutomationTests Tests/AutomationTimelineTests.cpp)
    target_include_directories(GrateAutomationTests PRIVATE Source)
    target_compile_features(GrateAutomationTests PRIVATE cxx_std_17)
    add_test(NAME GrateAutomationTests COMMAND GrateAutomationTests)
    add_executable(GrateTimingTests Tests/TimingTests.cpp)
    target_include_directories(GrateTimingTests PRIVATE Source)
    target_compile_features(GrateTimingTests PRIVATE cxx_std_17)
    add_test(NAME GrateTimingTests COMMAND GrateTimingTests)
    foreach(test IN ITEMS WetDucker WetEqualizer)
        add_executable(Grate${test}Tests Tests/${test}Tests.cpp)
        target_include_directories(Grate${test}Tests PRIVATE Source)
        target_compile_features(Grate${test}Tests PRIVATE cxx_std_17)
        add_test(NAME Grate${test}Tests COMMAND Grate${test}Tests)
    endforeach()
    if(WIN32 AND INHA_PLUGIN_BUILD_VST3)
        add_library(GrateEditorFixture MODULE ${GRATE_PLUGIN_SOURCES} Tests/EditorTestLicense.cpp)
        target_compile_features(GrateEditorFixture PRIVATE cxx_std_17)
        target_include_directories(GrateEditorFixture PRIVATE Source Dependencies/VST3_SDK)
        target_compile_definitions(GrateEditorFixture PRIVATE INHA_EDITOR_TEST_FIXTURE=1
            WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
        target_link_libraries(GrateEditorFixture PRIVATE GrateEngine INHA::Core
            gdi32 gdiplus msimg32 user32 kernel32 ole32 uuid shell32 comctl32 advapi32)
        set_target_properties(GrateEditorFixture PROPERTIES PREFIX "" OUTPUT_NAME "GrateEditorFixture"
            RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/editor-fixture/Contents/x86_64-win/$<0:>"
            LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/editor-fixture/Contents/x86_64-win/$<0:>")
        add_custom_command(TARGET GrateEditorFixture POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Resources"
                "${CMAKE_BINARY_DIR}/editor-fixture/Contents/Resources")
        if(MSVC)
            target_compile_options(GrateEditorFixture PRIVATE /W4 /permissive- /EHsc /utf-8)
        endif()
        add_dependencies(GrateEditorFixture INHA_Grate_VST3)
        foreach(test IN ITEMS ModuleSmoke EditorSmoke)
            add_executable(Grate${test} Tests/${test}.cpp)
            target_compile_features(Grate${test} PRIVATE cxx_std_17)
            target_include_directories(Grate${test} PRIVATE Source Dependencies/VST3_SDK)
            target_compile_definitions(Grate${test} PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
            target_link_libraries(Grate${test} PRIVATE gdi32 gdiplus user32 ole32 uuid shell32 comctl32)
            if(MSVC)
                target_compile_options(Grate${test} PRIVATE /W4 /permissive- /EHsc /utf-8)
            endif()
            add_dependencies(Grate${test} GrateEditorFixture)
        endforeach()
        add_test(NAME GrateModuleSmoke COMMAND GrateModuleSmoke "${VST3_BUNDLE}/Contents/x86_64-win/INHA Grate.vst3" "$<TARGET_FILE:GrateEditorFixture>")
        add_test(NAME GrateEditorSmoke COMMAND GrateEditorSmoke "$<TARGET_FILE:GrateEditorFixture>" "${CMAKE_BINARY_DIR}/editor-previews")
    endif()
endif()
